git init . 		-> Cria um repositório .git/ na pasta atual

git status	 	-> Mostra o status do trabalho (de cada 
			   arquivo)

git add 'nome_arquivo' 	-> Adiciona/prepara o arquivo 
			   para ser commitado 

git commit -m "msg qlqr" 	-> Faz commit do que estava/aguardava 
				   para ser comitado

git log 		-> Exibe histórico de commits

git reset 		-> Retorna para unstage

git reset idCommit 	-> Retorna para pos-commit
